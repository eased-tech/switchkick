import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid>
  <Card title="Clients work unchanged" icon="laptop">
    Codex, Claude Code and apps built on the OpenAI or Anthropic SDKs only change a base URL.
    switchkick accepts Chat Completions, Responses and Messages requests and streams the responses
    back byte for byte.
  </Card>
  <Card title="One key" icon="seti:lock">
    One `OPENROUTER_API_KEY` covers both the routing decision and the model call.
  </Card>
  <Card title="Three routing modes" icon="random">
    **static** sends everything to one tier. **hosted** lets OpenRouter's Jev router pick.
    **local** asks Jev and applies switchkick's own thresholds and guards.
  </Card>
  <Card title="No prompt text in logs" icon="document">
    The decision log records the tier, model, cost and Jev's answer, never what you or the model
    wrote. `switchkick report` sums it up.
  </Card>
</CardGrid>
